# Universum Infinitum - Character Traits
# Shared leader traits - Stellaris 4.5 (Cygnus) leader system
#
# "Ruler" and "admiral/general/governor" are no longer leader classes in 4.x, so the
# legacy ruler exclusion and class list are gone. This trait now applies to any
# commander, scientist or official that is not the empire ruler, and mirrors the
# vanilla starting-pool trait pattern (leader_trait_eager) it opposes.

ui_trait_leader_politically_connected = {
    inline_script = {
        script = trait/icon
        CLASS = leader
        ICON = "GFX_ui_trait_politically_connected"
        RARITY = common
        COUNCIL = no
        TIER = 1
    }

    leader_potential_add = {
        trait_is_crisis_faction_check = no
        is_ruler = no
        NOT = {
            from = {
                # TODO: Also meritocracy
                has_ethic = "ethic_gestalt_consciousness"
            }
        }
    }

    self_modifier = {
        leaders_cost_mult = -0.5
        species_leader_exp_gain = -0.25
    }

    leader_class = { commander scientist official }

    opposites = { "leader_trait_eager" }

    selectable_weight = {
        weight = @shared_trait_weight
        inline_script = paragon/not_council_weight_mult
    }
}
