# Universum Infinitum - Character Traits
# Ruler traits - Stellaris 4.5 (Cygnus) leader system
#
# The legacy "ruler" leader class no longer exists in 4.x. Rulers are councilors, so
# these traits are now modelled the way vanilla handles ruler traits: a councilor
# modifier that takes effect while the leader sits on the council (including as the
# empire ruler). Without Galactic Paragons they are open to any leader; with it they
# are reserved for the ruler and for leaders holding a councilor subclass.

ui_trait_ruler_backroom_backstabber = {
    inline_script = {
        script = trait/icon
        CLASS = leader
        ICON = "GFX_ui_trait_ruler_backroom_backstabber"
        RARITY = common
        COUNCIL = yes
        TIER = 1
    }

    leader_potential_add = {
        trait_is_crisis_faction_check = no
        NOT = {
            from = { has_authority = "auth_hive_mind" }
        }
        OR = {
            is_ruler = yes
            has_council_subclass = yes
            has_paragon_dlc = no
        }
    }

    councilor_modifier = {
        country_influence_produces_mult = 0.25
        pop_government_ethic_attraction = 0.05
    }

    leader_class = { commander scientist official }

    selectable_weight = {
        weight = @shared_trait_weight
        inline_script = paragon/council_weight_mult
    }
}
