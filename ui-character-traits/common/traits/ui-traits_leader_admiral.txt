# Universum Infinitum - Character Traits
# Admiral traits - Stellaris 4.5 (Cygnus) leader system
#
# The legacy "admiral" leader class no longer exists in 4.x; fleet commanders are
# now "commander" leaders with the "commander_admiral" subclass. These traits are
# therefore scoped to the commander class and, when the Galactic Paragons DLC is
# active, further restricted to the admiral subclass so they stay fleet-only.

ui_trait_admiral_berserker = {
    # Basic trait: remains attainable without Galactic Paragons, matching pre-4.x behaviour.
    inline_script = {
        script = trait/icon
        CLASS = commander
        ICON = "GFX_ui_trait_berserker"
        RARITY = common
        COUNCIL = no
        TIER = 1
    }
    leader_potential_add = {
        trait_is_crisis_faction_check = no
        OR = {
            has_paragon_dlc = no
            has_trait = subclass_commander_admiral
        }
    }
    fleet_modifier = {
        ship_weapon_damage = 0.15
        ship_disengage_chance_mult = -0.25
    }
    leader_class = { commander }
    opposites = {
        "leader_trait_nervous"
        "leader_trait_cautious"
        "leader_trait_scout"
        "ui_trait_admiral_guerillero"
    }
    selectable_weight = {
        weight = @class_trait_weight
        inline_script = paragon/pilot_weight_mult
        inline_script = {
            script = paragon/subclass_weight_mult
            SUBCLASS = commander_admiral
        }
    }
}
